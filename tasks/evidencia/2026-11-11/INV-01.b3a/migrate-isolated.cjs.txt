const fs = require('node:fs');
const { createHash } = require('node:crypto');
const { parse } = require('dotenv');
const { spawnSync } = require('node:child_process');
const tag = '0139_inv01b_application_scope';
const journal = JSON.parse(fs.readFileSync('drizzle/meta/_journal.json', 'utf8'));
if (journal.entries.at(-1).tag !== tag) throw new Error('Lote posterior sin revisar');
if (createHash('sha256').update(fs.readFileSync(`drizzle/${tag}.sql`)).digest('hex') !== 'adcd6dda1408584323d4444b9c82d606e00ddd94940b5a7124549f44311ab7f2') throw new Error('SQL distinto del revisado');
const isolated = parse(fs.readFileSync('.env.test')).TEST_DATABASE_URL;
if (new URL(isolated).hostname.replace(/-pooler(?=\.)/, '') !== 'ep-cold-morning-bewqoz3o.c-9.us-east-2.aws.neon.tech') throw new Error('Destino aislado inesperado');
const result = spawnSync('cmd.exe', ['/d', '/c', 'pnpm.cmd', 'db:migrate'], { stdio: 'inherit', env: {
  ...process.env, NODE_OPTIONS: '--use-system-ca --max-old-space-size=4096', DATABASE_URL: isolated, MIGRATION_DATABASE_URL: isolated,
} });
process.exit(result.status ?? 1);
